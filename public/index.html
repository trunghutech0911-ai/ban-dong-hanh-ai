Sổ Tay Công Thức Toán & Ngữ Pháp Tiếng Anh Trọng Tâm (Lớp 6 – Lớp 9)

📐 1. Công Thức Đại Số & Hình Học Cơ Bản

  • Lớp 6 – Tập hợp & Số nguyên: Quy tắc dấu ngoặc: Khi bỏ dấu ngoặc có dấu "-" đằng trước, ta phải đổi dấu tất cả các số hạng trong ngoặc.
  • Lớp 7 – Tỉ lệ thức: Nếu a/b = c/d thì a.d = b.c. Tính chất dãy tỉ số bằng nhau: a/b = c/d = (a+c)/(b+d).
  • Lớp 8 – 7 Hằng đẳng thức đáng nhớ:
    • Bình phương của một tổng: (A + B)² = A² + 2AB + B²
    • Bình phương của một hiệu: (A - B)² = A² - 2AB + B²
    • Hiệu hai bình phương: A² - B² = (A - B)(A + B)
  • Lớp 9 – Hệ thức lượng trong tam giác vuông: b² = a.b', h² = b'.c', Định lý Pytago: a² = b² + c².

📘 2. Ngữ Pháp Tiếng Anh THCS (Grades 6 - 9)

  • Present Simple (Hiện tại đơn): S + V(s/es) + O — Diễn tả thói quen hàng ngày hoặc sự thật hiển nhiên.
  • Present Continuous (Hiện tại tiếp diễn): S + am/is/are + V-ing — Diễn tả hành động đang xảy ra tại thời điểm nói.
  • Past Simple (Quá khứ đơn): S + V2/V-ed + O — Hành động đã hoàn tất trong quá khứ.
  • Conditional Sentences (Câu điều kiện):
    • Loại 1: If + S + V(htđ), S + will/can + V-inf
    • Loại 2: If + S + V(qkđ), S + would/could + V-inf

var DEFAULT_ALERT_EMAILS = ""; // VD: "hieutruong@truong.edu.vn, tamly@truong.edu.vn"

function getSheet_() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Mã Hồ Sơ", "Thời Gian", "Khối Lớp", "Chế Độ", "Học Sinh/Biệt Danh", "Phân Loại", "Mức Độ", "Luồng Zalo", "Địa Điểm", "Bằng Chứng / Liên Hệ", "Nội Dung", "Trạng Thái", "Phản Hồi GV", "Người Phản Hồi", "JSON_SYNC"]);
  }
  return sheet;
}

function doPost(e) {
  var sheet = getSheet_();
  var data = JSON.parse(e.postData.contents || "{}");
  var r = data.row || {};
  var rawJson = JSON.stringify(data.rawRecord || {});
  var rowVals = [r.trackingCode || "", r.createdAt || "", r.grade || "", r.identityMode || "", r.studentDisplay || "", r.categoryLabel || "", r.urgency || "", r.zaloChannels || "", r.locationAndTime || "", r.evidenceAndContact || "", r.description || "", r.status || "", r.counselorReply || "", r.repliedBy || "", rawJson];

  // UPSERT theo Mã Hồ Sơ (Cột A) để không tạo dòng trùng khi GV phản hồi hoặc HS nhắn nối tiếp
  var lastRow = sheet.getLastRow();
  var targetRow = -1;
  if (lastRow > 1 && r.trackingCode) {
    var codes = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
    for (var i = 0; i < codes.length; i++) {
      if (String(codes[i][0]).toUpperCase() === String(r.trackingCode).toUpperCase()) {
        targetRow = i + 2;
        break;
      }
    }
  }
  if (targetRow > 0) {
    sheet.getRange(targetRow, 1, 1, rowVals.length).setValues([rowVals]);
  } else {
    sheet.appendRow(rowVals);
    targetRow = sheet.getLastRow();
  }

  // TỰ ĐỘNG GỬI EMAIL CẢNH BÁO KHẨN TỨC THÌ (BƯỚC 3B: HIGH / CRITICAL)
  var emailSent = false;
  var recipients = (data.alertEmails || DEFAULT_ALERT_EMAILS || "").trim();
  var urg = String(r.urgency || "").toUpperCase();
  if (data.isNew && recipients && (urg === "HIGH" || urg === "CRITICAL")) {
    var subject = "🚨 [KHẨN CẤP BƯỚC 3B] Hồ sơ " + r.trackingCode + " - " + (r.categoryLabel || "Cảnh báo an toàn học sinh");
    var body = "HỆ THỐNG BẠN ĐỒNG HÀNH AI PHÁT HIỆN CA NGUY CƠ CAO CẦN CAN THIỆP NGAY:\n\n"
      + "• Mã hồ sơ: " + r.trackingCode + "\n"
      + "• Học sinh: " + r.studentDisplay + " (" + r.grade + " - " + r.identityMode + ")\n"
      + "• Phân loại: " + r.categoryLabel + " | Mức độ: " + urg + "\n"
      + "• Luồng Zalo định tuyến: " + r.zaloChannels + "\n"
      + "• Địa điểm/Thời gian: " + r.locationAndTime + "\n"
      + "• Bằng chứng/Liên hệ: " + (r.evidenceAndContact || "Không có") + "\n"
      + "• Nội dung học sinh phản ánh:\n\"" + r.description + "\"\n\n"
      + "----------------------------------------\n"
      + "🧠 THAM MƯU XỬ LÝ TỪ TRỢ LÝ AI:\n" + (data.aiCoPilotSummary || "Ưu tiên bảo vệ an toàn học sinh ngay lập tức, không gọi đối chất trực tiếp.");
    MailApp.sendEmail(recipients, subject, body);
    emailSent = true;
  }

  return ContentService.createTextOutput(JSON.stringify({ ok: true, rowNumber: targetRow, emailSent: emailSent })).setMimeType(ContentService.MimeType.JSON);
}

// ĐỌC NGƯỢC 2 CHIỀU (LIST / LOOKUP) ĐỂ KHÔNG BAO GIỜ MẤT MÃ TRA CỨU NA-XXXX KHI VERCEL NGỦ ĐÔNG
function doGet(e) {
  var sheet = getSheet_();
  var action = (e && e.parameter && e.parameter.action) ? String(e.parameter.action).toUpperCase() : "LIST";
  var code = (e && e.parameter && e.parameter.code) ? String(e.parameter.code).toUpperCase() : "";
  var lastRow = sheet.getLastRow();
  var results = [];
  if (lastRow > 1) {
    var rows = sheet.getRange(2, 1, lastRow - 1, 15).getValues();
    for (var i = rows.length - 1; i >= 0; i--) {
      var jsonStr = rows[i][14];
      if (!jsonStr) continue;
      try {
        var obj = JSON.parse(jsonStr);
        if (action === "LOOKUP") {
          if (String(obj.trackingCode || "").toUpperCase() === code) {
            return ContentService.createTextOutput(JSON.stringify({ ok: true, request: obj })).setMimeType(ContentService.MimeType.JSON);
          }
        } else {
          results.push(obj);
        }
      } catch (_err) {}
    }
  }
  if (action === "LOOKUP") {
    return ContentService.createTextOutput(JSON.stringify({ ok: false, error: "Not found" })).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify({ ok: true, requests: results })).setMimeType(ContentService.MimeType.JSON);
}

Mục Tiêu Cốt Lõi & Sáu Chức Năng Chính (Mục 1 & Mục 3)

Ý tưởng cốt lõi: Xây dựng một ứng dụng sử dụng trí tuệ nhân tạo (AI), giúp học sinh lớp 6–9 có nơi chia sẻ những khó khăn tâm lý, được hướng dẫn xử lý tình huống bắt nạt học đường và kết nối với người hỗ trợ phù hợp mà không bắt buộc tiết lộ danh tính khi bắt đầu trò chuyện.

🎯 1. Năm Mục Tiêu Của Ứng Dụng
  • Tạo không gian an toàn, thân thiện để học sinh chia sẻ cảm xúc.
  • Phát hiện sớm những dấu hiệu học sinh đang bị bắt nạt, cô lập hoặc đe dọa.
  • Cá thể hóa lời khuyên theo độ tuổi (Khối 6–9), cảm xúc và tình huống của từng học sinh.
  • Hướng dẫn kỹ năng tự bảo vệ, ứng xử không bạo lực và tìm kiếm sự giúp đỡ.
  • Hỗ trợ giáo viên phụ trách tư vấn học đường tiếp nhận và xử lý những tình huống cần can thiệp.
⚙️ 3. Sáu Chức Năng Chính Đã Tích Hợp
  • 3.1. AI tư vấn tâm lý ẩn danh: Chọn biệt danh, không cần tài khoản; AI lắng nghe, đặt câu hỏi ngắn dễ hiểu, không tự chẩn đoán bệnh tâm lý.
  • 3.2. Cá thể hóa hỗ trợ: Theo khối lớp (6–9), cảm xúc, vai trò (người bị bắt nạt, người chứng kiến, người có hành vi bắt nạt) và mức độ an toàn.
  • 3.3. Hộp thư báo cáo bắt nạt: Văn bản, nhóm tình huống, bổ sung bằng chứng khi cần; chọn ẩn danh hoặc để lại liên hệ.
  • 3.4. Thư viện kỹ năng phòng chống bắt nạt: Tình huống tương tác (lời nói, thể chất, cô lập, trên mạng, chứng kiến, sửa sai).
  • 3.5. Kết nối giáo viên tư vấn: Ưu tiên chuyển ca nguy cơ cao theo quy trình bảo vệ trẻ em.
  • 3.6. Thống kê và cảnh báo cho nhà trường: Tổng hợp xu hướng ở quy mô lớn, không làm lộ danh tính học sinh.

Bước 3B: Có nguy cơ mất an toàn
AI ưu tiên các bước đảm bảo an toàn, hướng dẫn liên hệ người lớn (111) và chuyển thông tin theo quy trình can thiệp.
💡 Ví dụ tình huống sử dụng thực tế: Khi một học sinh cho biết thường xuyên bị bạn bè chế giễu trong nhóm lớp, AI không chỉ an ủi chung chung mà còn hướng dẫn lưu bằng chứng, tránh tranh cãi có thể gây nguy hiểm và chọn một giáo viên đáng tin cậy để báo cáo. Nếu học sinh đang bị đe dọa trực tiếp, ứng dụng ưu tiên hướng dẫn tìm nơi an toàn và người hỗ trợ ngay lập tức.

4 Nguyên Tắc An Toàn & Đạo Đức AI Học Đường

  • Ẩn danh có giới hạn: Không hứa giữ bí mật tuyệt đối. Ứng dụng giải thích rõ những trường hợp phải chia sẻ thông tin tối thiểu cần thiết nhằm bảo vệ học sinh trước nguy cơ nghiêm trọng.
  • Giám sát bởi con người: AI không thay thế chuyên viên tâm lý, không tự kết luận ai có lỗi và không tự quyết định kỷ luật.
  • Bảo vệ dữ liệu: Hạn chế thu thập và lưu hội thoại, mã hóa dữ liệu (SHA-256), phân quyền truy cập (RBAC), quy định thời hạn xóa và tuân thủ quy định về dữ liệu cá nhân trẻ em.
  • Can thiệp khẩn cấp: Khi có dấu hiệu tự gây hại, bạo lực nghiêm trọng hoặc xâm hại, hướng dẫn học sinh tìm người lớn hỗ trợ ngay và liên hệ Tổng đài Quốc gia bảo vệ trẻ em 111 (miễn phí, 24/7).
